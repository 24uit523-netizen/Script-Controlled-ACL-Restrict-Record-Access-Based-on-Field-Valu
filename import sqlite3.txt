import sqlite3
import hashlib
from datetime import datetime
from getpass import getpass


# ============================================================
# DATABASE CONFIGURATION
# ============================================================

DB_NAME = "acl_system.db"


def get_connection():
    conn = sqlite3.connect(DB_NAME)
    conn.row_factory = sqlite3.Row
    return conn


# ============================================================
# PASSWORD HASHING
# ============================================================

def hash_password(password):
    return hashlib.sha256(password.encode()).hexdigest()


# ============================================================
# DATABASE INITIALIZATION
# ============================================================

def initialize_database():
    conn = get_connection()
    cursor = conn.cursor()

    # Users table
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password TEXT NOT NULL,
            department TEXT NOT NULL,
            role TEXT NOT NULL DEFAULT 'user'
        )
    """)

    # Records table
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS records (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            title TEXT NOT NULL,
            description TEXT,
            department TEXT NOT NULL,
            created_by INTEGER,
            created_at TEXT NOT NULL,
            updated_at TEXT NOT NULL,
            FOREIGN KEY (created_by) REFERENCES users(id)
        )
    """)

    # Audit log
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS audit_logs (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT,
            action TEXT,
            record_id INTEGER,
            result TEXT,
            timestamp TEXT NOT NULL
        )
    """)

    conn.commit()

    # Create default users
    create_default_users(conn)

    # Create sample records
    create_sample_records(conn)

    conn.close()

# AUDIT LOG
# ============================================================

def log_action(username, action, record_id, result):
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        INSERT INTO audit_logs
        (username, action, record_id, result, timestamp)
        VALUES (?, ?, ?, ?, ?)
    """, (
        username,
        action,
        record_id,
        result,
        datetime.now().isoformat()
    ))

    conn.commit()
    conn.close()

# LOGIN

def login():
    print("\n" + "=" * 60)
    print("              SECURE RECORD ACCESS SYSTEM")
    print("=" * 60)

    username = input("Username: ")
    password = getpass("Password: ")

    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT *
        FROM users
        WHERE username = ?
        AND password = ?
    """, (username, hash_password(password)))

    user = cursor.fetchone()

    conn.close()

    if user:
        print("\nLogin successful!")
        print(f"Welcome, {user['username']}")
        print(f"Department: {user['department']}")
        print(f"Role: {user['role']}")

        return user

    print("\nInvalid username or password.")
    return None


# ============================================================
# ACCESS CONTROL LIST
# ============================================================

def check_access(user, record, action="read"):
    """
    Main ACL function.

    Rules:
        1. Admin -> Full access
        2. Normal user -> Access only if
           user's department == record's department
        3. Unknown user -> Denied
    """

    # --------------------------------------------------------
    # Rule 1: Admin Override
    # --------------------------------------------------------

    if user["role"] == "admin":

        log_action(
            user["username"],
            action,
            record["id"],
            "ALLOWED - ADMIN"
        )

        return True

    # --------------------------------------------------------
    # Rule 2: Department Based Access
    # --------------------------------------------------------

    if user["department"] == record["department"]:

        log_action(
            user["username"],
            action,
            record["id"],
            "ALLOWED - DEPARTMENT MATCH"
        )

        return True

    # --------------------------------------------------------
    # Rule 3: Access Denied
    # --------------------------------------------------------

    log_action(
        user["username"],
        action,
        record["id"],
        "DENIED - DEPARTMENT MISMATCH"
    )

    return False

# ============================================================
# VIEW RECORDS
# ============================================================

def view_records(user):
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT *
        FROM records
        ORDER BY id
    """)

    records = cursor.fetchall()

    conn.close()

    print("\n" + "=" * 60)
    print("                  AVAILABLE RECORDS")
    print("=" * 60)

    found = False

    for record in records:

        if check_access(user, record, "READ"):

            display_record(record)
            found = True

        else:

            print(
                f"\nRecord {record['id']} - "
                f"ACCESS DENIED"
            )

    if not found:
        print("\nNo records available for your department.")

def view_single_record(user):
    try:
        record_id = int(input("\nEnter Record ID: "))
    except ValueError:
        print("Invalid ID.")
        return

    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT *
        FROM records
        WHERE id = ?
    """, (record_id,))

    record = cursor.fetchone()

    conn.close()

    if record is None:
        print("Record not found.")
        return

    if check_access(user, record, "READ"):
        display_record(record)
    else:
        print("\nACCESS DENIED")
        print(
            "You cannot access records "
            "from another department."
        )


# ============================================================
# CREATE RECORD
# ============================================================

def create_record(user):

    print("\n" + "=" * 60)
    print("                    CREATE RECORD")
    print("=" * 60)

    title = input("Title: ")
    description = input("Description: ")

    # Normal users can create only for their department
    if user["role"] == "admin":

        print("\nDepartments:")
        print("1. IT")
        print("2. HR")
        print("3. Finance")

        choice = input("Select department: ")

        departments = {
            "1": "IT",
            "2": "HR",
            "3": "Finance"
        }

        department = departments.get(choice)

        if department is None:
            print("Invalid department.")
            return

    else:

        department = user["department"]

    current_time = datetime.now().isoformat()

    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        INSERT INTO records
        (title, description, department, created_by,
         created_at, updated_at)
        VALUES (?, ?, ?, ?, ?, ?)
    """, (
        title,
        description,
        department,
        user["id"],
        current_time,
        current_time
    ))

    record_id = cursor.lastrowid

    conn.commit()
    conn.close()

    log_action(
        user["username"],
        "CREATE",
        record_id,
        "SUCCESS"
    )

    print("\nRecord created successfully.")
    print(f"Record ID: {record_id}")


# ============================================================
# UPDATE RECORD
# ============================================================

def update_record(user):

    try:
        record_id = int(input("\nEnter Record ID: "))
    except ValueError:
        print("Invalid ID.")
        return

    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT *
        FROM records
        WHERE id = ?
    """, (record_id,))

    record = cursor.fetchone()

    if record is None:
        conn.close()
        print("Record not found.")
        return

    # ACL check
    if not check_access(user, record, "UPDATE"):

        conn.close()

        print("\nUPDATE ACCESS DENIED")
        return

    print("\nCurrent Record:")
    display_record(record)

    new_title = input(
        "New title (press Enter to keep old): "
    )

    new_description = input(
        "New description (press Enter to keep old): "
    )

    if new_title == "":
        new_title = record["title"]

    if new_description == "":
        new_description = record["description"]

    cursor.execute("""
        UPDATE records
        SET title = ?,
            description = ?,
            updated_at = ?
        WHERE id = ?
    """, (
        new_title,
        new_description,
        datetime.now().isoformat(),
        record_id
    ))

    conn.commit()
    conn.close()

    log_action(
        user["username"],
        "UPDATE",
        record_id,
        "SUCCESS"
    )

    print("\nRecord updated successfully.")


# ============================================================
# DELETE RECORD
# ============================================================

def delete_record(user):

    try:
        record_id = int(input("\nEnter Record ID: "))
    except ValueError:
        print("Invalid ID.")
        return

    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT *
        FROM records
        WHERE id = ?
    """, (record_id,))

    record = cursor.fetchone()

    if record is None:
        conn.close()
        print("Record not found.")
        return

    # ACL check
    if not check_access(user, record, "DELETE"):

        conn.close()

        print("\nDELETE ACCESS DENIED")
        return

    confirmation = input(
        "Are you sure you want to delete this record? (yes/no): "
    )

    if confirmation.lower() != "yes":

        conn.close()

        print("Delete operation cancelled.")
        return

    cursor.execute("""
        DELETE FROM records
        WHERE id = ?
    """, (record_id,))

    conn.commit()
    conn.close()

    log_action(
        user["username"],
        "DELETE",
        record_id,
        "SUCCESS"
    )

    print("\nRecord deleted successfully.")

        SELECT *

    conn.close()

    found = False

    print("\n" + "=" * 60)
    print("                    SEARCH RESULTS")
    print("=" * 60)

    for record in records:

        if check_access(user, record, "SEARCH"):

            display_record(record)
            found = True

    if not found:
        print("\nNo accessible records found.")

        print("1. View All Accessible Records")
        print("2. View Single Record")
        print("3. Create Record")
        print("4. Update Record")
        print("5. Delete Record")
        print("6. Search Records")
        print("7. My Profile")

        if user["role"] == "admin":
            print("8. View Users")
            print("9. View Audit Logs")
            print("10. Logout")
        else:
            print("8. Logout")

        choice = input("\nEnter your choice: ")

        # ----------------------------------------------------
        # USER OPTIONS
        # ----------------------------------------------------

        if choice == "1":
            view_records(user)

        elif choice == "2":
            view_single_record(user)

        elif choice == "3":
            create_record(user)

        elif choice == "4":
            update_record(user)

        elif choice == "5":
            delete_record(user)

        elif choice == "6":
            search_records(user)

        elif choice == "7":
            show_user_information(user)

        # ----------------------------------------------------
        # ADMIN OPTIONS
        # ----------------------------------------------------

        elif choice == "8" and user["role"] == "admin":
            view_users(user)

        elif choice == "9" and user["role"] == "admin":
            view_audit_logs(user)

        # ----------------------------------------------------
        # LOGOUT
        # ----------------------------------------------------

        elif (
            (choice == "10" and user["role"] == "admin")
            or
            (choice == "8" and user["role"] != "admin")
        ):

            print("\nLogged out successfully.")
            break

        else:

            print("\nInvalid choice.")
